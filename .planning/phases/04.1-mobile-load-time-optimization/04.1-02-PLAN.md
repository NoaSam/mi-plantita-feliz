 * Return a thumbnail URL for a Supabase Storage image, or the original URL if:
 *  - The URL is a data: URI (legacy, pass-through)
 *  - The URL is not a Supabase Storage /object/public/ URL

 * @param width Target width in pixels (default 200). Retina displays should pass 2x display size.

    ```typescript
    /**
     * Thumbnail URL helper for Supabase Storage images.
     *
     * D-08 asks for on-the-fly `?width=200&quality=70` transformations. That
     * feature is Pro-plan only (Free plan returns 400/404 on the transform
     * endpoint). Per CPO CRITICAL DECISION #1 (planner locked): the project
     * stays on Free plan. This helper defaults to serving originals safely
     * and auto-upgrades if Pro is later activated.
     *
     * Strategy:
     *  1. First call kicks off a HEAD request to a well-formed transform URL.
     *  2. While HEAD is in-flight, return ORIGINAL URL (never block render).
     *  3. HEAD resolves 200 → cache tier="available" → future calls return transformed URL.
     *  4. HEAD resolves 4xx / network error → cache tier="unavailable" → future calls keep returning originals.
     *  5. Cache persists in-memory for the app lifetime (module-scope).
     */

    type Tier = "unknown" | "detecting" | "available" | "unavailable";
    let tier: Tier = "unknown";

    const OBJECT_PATH = "/storage/v1/object/public/";
    const RENDER_PATH = "/storage/v1/render/image/public/";

    /** True if url is a Supabase Storage /object/public/ URL. */
    function isSupabaseStorageUrl(url: string): boolean {
      return typeof url === "string" && url.includes(OBJECT_PATH);
    }

    /** Rewrite object → render URL and append width/quality. */
    function rewriteToTransform(url: string, width: number, quality: number): string {
      const rendered = url.replace(OBJECT_PATH, RENDER_PATH);
      const sep = rendered.includes("?") ? "&" : "?";
      return `${rendered}${sep}width=${width}&quality=${quality}`;
    }

    /** Fire-and-forget HEAD to probe transform availability. */
    function detectTier(sampleUrl: string): void {
      if (tier !== "unknown") return;
      tier = "detecting";
      const probeUrl = rewriteToTransform(sampleUrl, 100, 70);
      // fetch() may reject on network error → catch to swallow.
      fetch(probeUrl, { method: "HEAD" })
        .then((res) => {
          tier = res.ok ? "available" : "unavailable";
        })
        .catch(() => {
          tier = "unavailable";
        });
    }

    /** Public helper. */
    export function getThumbnailUrl(url: string, width = 200, quality = 70): string {
      if (!url) return url;
      if (typeof url !== "string") return url;
      if (url.startsWith("data:")) return url;              // legacy pass-through
      if (!isSupabaseStorageUrl(url)) return url;           // non-Supabase pass-through

      if (tier === "unknown") {
        detectTier(url);
        return url;                                          // never block first render
      }
      if (tier === "available") {
        return rewriteToTransform(url, width, quality);
      }
      // "detecting" or "unavailable" → serve original
      return url;
    }

    /** Test-only: reset module cache between tests. */
    export function resetThumbnailTierCacheForTests(): void {
      tier = "unknown";
    }
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → Supabase Storage HTTPS (`getPublicUrl` responses) | User-facing image URL requests, cross-origin |
| Browser → OSM tile CDN (osmfoundation.org) | Cross-origin GET for map tiles |
| Browser → Supabase render/image endpoint (HEAD probe) | One-time HEAD from thumbnail helper |
| String composition of image URLs | User's own image URL data crosses into <img src> — must not enable injection |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04.1.02-01 | Tampering | Cache-poisoning via URL query params | mitigate | Helper builds transform URLs from OUR own template + hard-coded width/quality numeric params. No user input flows into path. `sep` is either `?` or `&` — cannot escape URL structure. |
| T-04.1.02-02 | Information Disclosure | Cross-origin image request leaks referrer | accept | Default browser referrer-policy leaks origin only, not full path. Photos are already public (bucket `public=true` per Phase 1). Adding `referrerpolicy="no-referrer"` could be considered for defense-in-depth but not required by ASVS L1. |
| T-04.1.02-03 | Injection (XSS) | User-controlled URL fragment reaches `<img src>` | mitigate | React auto-escapes attribute values via JSX. Helper never wraps URL in template strings that end up in `dangerouslySetInnerHTML`. `<img src={...}>` uses attribute quoting. No XSS surface. |
| T-04.1.02-04 | Denial of Service | Infinite retry of tier-detect HEAD on 500 errors | accept | Detection is one-shot: after `tier` transitions from "unknown"/"detecting" to "available"/"unavailable" it stays there for app lifetime. Even on network error, `catch` sets `tier="unavailable"` — no retry loop. |
| T-04.1.02-05 | Information Disclosure | Preconnect leaks user's IP + browser fingerprint to OSM/Supabase even if the app is idle | accept | Both hosts are ALREADY contacted the moment the app loads any page (Supabase auth check, OSM tiles on /mapa). Preconnect doesn't disclose new information. |
| T-04.1.02-06 | Spoofing | Attacker-controlled URL that looks like `/storage/v1/object/public/` but is not Supabase | mitigate | Helper only rewrites paths, never re-hosts requests. The domain in the URL is preserved. If URL is `attacker.com/storage/v1/object/public/x.jpg` the transform URL becomes `attacker.com/storage/v1/render/image/public/x.jpg?width=200` — request still goes to attacker.com. Since imageUrl comes from OUR DB (with backfill controlled by us), attacker cannot inject arbitrary URLs into plant_searches.image_url without RLS bypass. |
| T-04.1.02-07 | Repudiation | No audit of image URL rewrites | accept | Not applicable — rewriting client-side URLs is not a security-material action requiring audit. |

**Note:** The one-time HEAD probe uses whatever user's session is active. On Free plan it will return 400; on Pro it will return 200. Either way, no sensitive state changes.
</threat_model>
